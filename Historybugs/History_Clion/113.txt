class Human {
    float height;
    float weight;
    int age;
    std::string name;
public:
    Human(float height, float weight, int age);
    float getHeight() const {
        return height;
    }
    void setHeight(float height) {
        Human::height = height;
    }
    float getWeight() const {
        return weight;
    }
    void setWeight(float weight) {
        Human::weight = weight;
    }
    int getAge() const {
        return age;
    }
    void makeSound(int type) {};
    float bmi();
};

// Human.cpp
Human::Human(float height, float weight, int age)
    : height(height), weight(weight), age(age)
{
}

float Human::bmi() {
    if (height != 0) {
        return weight / ((height * height));
    }
    return 0;
}
