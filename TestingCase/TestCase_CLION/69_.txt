public :

};
typedef void (Foo::*FOO_call)();

Foo pfoo = Foo();
FOO_call pfnTempCall = (FOO_call)(&Foo::method);
Foo* ptrObj = foo; //{Extract variable for Foo pointer instance}

(pfoo.*pfnTempCall)();

}

{
