package main
import (
"fmt"
)
type Foo interface {
Frob(int) string
}
type fooImpl struct {
frob func(int) string
}
func (f *fooImpl) Frob(i int) string {
return f.frob(i)
}
func NewFoo(frob func(int) string) Foo {
return &fooImpl{frob: frob}
}
func buildFooHex() Foo {
return NewFoo(func(i int) string {
return fmt.Sprintf("%x", i)
})
}
func main() {
foos := []Foo{
buildFooHex(), // extract slice element into buildFooHex
NewFoo(func(i int) string {
return fmt.Sprintf("%d", i)
}),
}
for i := 0; i < 3; i++ {
for _, f := range foos {
println(f.Frob(i))
}
}
}
