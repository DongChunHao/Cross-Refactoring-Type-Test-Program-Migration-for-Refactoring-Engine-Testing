func finder(mine [5]string) {
var r []string
collectOre(mine, &r) // extract for loop logic into collectOre
return r
}
func collectOre(mine [5]string, r *[]string) {
for _, v := range mine {
if v == "ore" {
*r = append(*r, v)
}
}
}
