#include <vector>

struct Quadrilateral {
    Quadrilateral(int a, int b, int c, int d) {}
};
struct Triangle {
    int n1() {return 0;}
    int n2() {return 0;}
    int n3() {return 0;}
};

void demo()
{
    std::vector<Quadrilateral> quadrilaterals;
    Triangle* triangle = nullptr;
    int centerAB = 0, centerBC = 0, centerCA = 0, center = 0;

    quadrilaterals.push_back(Quadrilateral(centerAB, triangle->n2(), centerBC, center));
    quadrilaterals.push_back(Quadrilateral(centerBC, triangle->n3(), centerCA, center));
    quadrilaterals.push_back(Quadrilateral(centerCA, triangle->n1(), centerAB, center));
}
