package main

import (
	"fmt"
)

func Print(i interface{}) {
	switch o := i.(type) {
	case int64:
		fmt.Printf("%5d\n", o)
	case float64:
		fmt.Printf("%7.3f\n", o)
	case string:
		fmt.Printf("%s\n", o)
	default: // covers structs and such
		fmt.Printf("%+v\n", o)
	}
}

func main() {
	var a float64
	a = 1.33343
	Print(a)
}
