private async ValueTask<Guid[]> getHistoryAsync(DataSourceInfo source)
{
    await using (var factory = await this.DbContextFactoryCreator.CreateFactoryAsync(source).ConfigureAwait(false))

        {
            return ctx.Exports.OrderBy(it => it.Id).Select(it => it.Signature).ToArray();
        }
    }
}
