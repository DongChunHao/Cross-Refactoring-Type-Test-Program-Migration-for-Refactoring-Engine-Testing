// Complete compilable test code
type State = {
  entities: Record<string, { children: string[] }>;
};
declare const entityAdapter: {
  updateOne: <T>(opts: { id: string; changes: Partial<T> }, state: T) => T;
};

const updateParent = (state: State, parentName: string, name: string): State => {
    const parent = state.entities[parentName]
    return entityAdapter.updateOne({
        id: parentName,
        changes: {
            children: parent.children.concat(name)
        }
    }, state)
}
