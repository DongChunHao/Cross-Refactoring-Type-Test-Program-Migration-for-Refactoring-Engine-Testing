Refactoring new name: N/A

#include <iostream>
#include <string>
class BasePrinter {
public:
virtual void print(const std::string& key, int count) {}
};
class Printer : public BasePrinter //{Push down print implementation from BasePrinter to Printer}
{
public:
void print(const std::string& key, int count) override {
std::cout << count << "|" << key << "\n";
}
};
int main()
{
std::string key = "test_key";
int count = 10;
Printer p;
p.print(key, count);
return 0;
}
