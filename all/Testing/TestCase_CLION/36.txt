#include <iostream>
class BaseIO {
public:
virtual void print(const char *string);
};
class IOUtil : public BaseIO {
public:
void print(const char *string) override { std::cout << string; } //{Push down method print from BaseIO}
};

IOUtil util;

return 0;
}
