class BaseGenerator
{

};

{
public:

void writeCodeBody(std::ostream& hppout, std::ostream& cppout) override;
};
void AbstractGenerator::someFunc(std::ostream& hppout, std::ostream& cppout)
{

hppout << std::endl;
}
writeCodeBody(hppout, cppout);
}
