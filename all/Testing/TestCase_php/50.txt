```
<?php
class NodeBase { }
class HtmlNode extends NodeBase {
    protected static function escapeAttr(string $v): string { return htmlspecialchars($v, ENT_QUOTES); }
    function renderAttr(string $value)
    {
        if ($value !== "") {
            return HtmlNode::escapeAttr($value);
        }
        return "";
    }
}
```
