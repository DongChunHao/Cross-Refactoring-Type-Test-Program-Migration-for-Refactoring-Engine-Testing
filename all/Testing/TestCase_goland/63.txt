package jetbrainsbug
import "testing"
func Sum(a int, b int) int {
return a + b
}
func TestSum(t *testing.T) {
type args struct {
a int
b int
}
tests := []struct {
name string
args args
want int
}{
{
name: "test",
args: args{
a: 0,
b: 0,
},
want: 0,
},
}
for _, tt := range tests {
t.Run(tt.name, func(t *testing.T) {
runSum(tt.args.a, tt.args.b) // extract Sum call into runSum
})
}
}
func runSum(a, b int) {
Sum(a, b)
}
